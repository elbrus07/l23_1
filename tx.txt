#include <gtk/gtk.h>
#include <pango/pango.h>
#include <fstream>
#include <sstream>
#include <string>

using namespace std;

GtkWidget* editorView;

void applySyntaxHighlighting() {
    GtkTextBuffer* textBuffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));
    GtkTextIter startPos, endPos;

    gtk_text_buffer_get_start_iter(textBuffer, &startPos);
    gtk_text_buffer_get_end_iter(textBuffer, &endPos);

    GtkTextTagTable* tagTable = gtk_text_buffer_get_tag_table(textBuffer);
    GtkTextTag* tagBlue = gtk_text_tag_table_lookup(tagTable, "kw_blue");
    GtkTextTag* tagGreen = gtk_text_tag_table_lookup(tagTable, "kw_green");
    GtkTextTag* tagRed = gtk_text_tag_table_lookup(tagTable, "kw_red");

    if (tagBlue) gtk_text_buffer_remove_tag(textBuffer, tagBlue, &startPos, &endPos);
    if (tagGreen) gtk_text_buffer_remove_tag(textBuffer, tagGreen, &startPos, &endPos);
    if (tagRed) gtk_text_buffer_remove_tag(textBuffer, tagRed, &startPos, &endPos);

    gchar* rawText = gtk_text_buffer_get_text(textBuffer, &startPos, &endPos, FALSE);
    string textContent = rawText;

    auto applyKeywordTag = [&](const string& keyword, GtkTextTag* tag) {
        size_t foundPos = textContent.find(keyword);
        while (foundPos != string::npos) {
            bool isLeftBoundary = (foundPos == 0 || !g_ascii_isalnum(textContent[foundPos - 1]));
            bool isRightBoundary = (foundPos + keyword.length() == textContent.length() || !g_ascii_isalnum(textContent[foundPos + keyword.length()]));

            if (isLeftBoundary && isRightBoundary && tag) {
                GtkTextIter wordStart, wordEnd;
                gtk_text_buffer_get_iter_at_offset(textBuffer, &wordStart, foundPos);
                gtk_text_buffer_get_iter_at_offset(textBuffer, &wordEnd, foundPos + keyword.length());
                gtk_text_buffer_apply_tag(textBuffer, tag, &wordStart, &wordEnd);
            }
            foundPos = textContent.find(keyword, foundPos + 1);
        }
    };

    applyKeywordTag("int", tagBlue);
    applyKeywordTag("return", tagRed);
    applyKeywordTag("if", tagGreen);

    g_free(rawText);
}

void newFile(GtkWidget* widget, gpointer data) {
    GtkTextBuffer* buffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));
    gtk_text_buffer_set_text(buffer, "", -1);
}

void saveCurrentFile(GtkWidget* widget, gpointer data) {
    GtkTextBuffer* buffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));
    GtkTextIter start, end;
    gtk_text_buffer_get_start_iter(buffer, &start);
    gtk_text_buffer_get_end_iter(buffer, &end);

    gchar* text = gtk_text_buffer_get_text(buffer, &start, &end, FALSE);
    ofstream outFile("note.txt");
    if (outFile.is_open()) {
        outFile << text;
    }
    g_free(text);
}

void loadFile(GtkWidget* widget, gpointer data) {
    ifstream inFile("note.txt");
    if (!inFile.is_open()) return;

    stringstream fileContentStream;
    fileContentStream << inFile.rdbuf();

    GtkTextBuffer* editorBuffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));
    
    g_signal_handlers_block_by_func(editorBuffer, (gpointer)+[](GtkTextBuffer*, gpointer){}, NULL);
    
    gtk_text_buffer_set_text(editorBuffer, fileContentStream.str().c_str(), -1);
    applySyntaxHighlighting();
    
    g_signal_handlers_unblock_by_func(editorBuffer, (gpointer)+[](GtkTextBuffer*, gpointer){}, NULL);
}

void applyTextFormat(const gchar* formatName) {
    GtkTextBuffer* buffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));
    GtkTextIter startSel, endSel;

    if (gtk_text_buffer_get_selection_bounds(buffer, &startSel, &endSel)) {
        GtkTextTagTable* table = gtk_text_buffer_get_tag_table(buffer);
        GtkTextTag* formatTag = gtk_text_tag_table_lookup(table, formatName);

        if (formatTag) {
            if (gtk_text_iter_has_tag(&startSel, formatTag) && gtk_text_iter_has_tag(&endSel, formatTag)) {
                gtk_text_buffer_remove_tag(buffer, formatTag, &startSel, &endSel);
            } else {
                gtk_text_buffer_apply_tag(buffer, formatTag, &startSel, &endSel);
            }
        }
    }
}

void onBoldClicked(GtkWidget* widget, gpointer data) { applyTextFormat("fmt_bold"); }
void onItalicClicked(GtkWidget* widget, gpointer data) { applyTextFormat("fmt_italic"); }
void onUnderlineClicked(GtkWidget* widget, gpointer data) { applyTextFormat("fmt_underline"); }

void onApplicationActivate(GtkApplication* application, gpointer user_data) {
    GtkWidget* mainWindow = gtk_application_window_new(application);
    gtk_window_set_title(GTK_WINDOW(mainWindow), "Текстовый редактор (GTK4)");
    gtk_window_set_default_size(GTK_WINDOW(mainWindow), 900, 600);

    GtkWidget* mainVBox = gtk_box_new(GTK_ORIENTATION_VERTICAL, 5);
    gtk_window_set_child(GTK_WINDOW(mainWindow), mainVBox);

    GtkWidget* toolbarHBox = gtk_box_new(GTK_ORIENTATION_HORIZONTAL, 5);
    gtk_box_append(GTK_BOX(mainVBox), toolbarHBox);

    GtkWidget* newBtn = gtk_button_new_with_label("Новый");
    GtkWidget* openBtn = gtk_button_new_with_label("Открыть...");
    GtkWidget* saveBtn = gtk_button_new_with_label("Сохранить...");
    
    GtkWidget* boldBtn = gtk_button_new_with_label("B");
    GtkWidget* italicBtn = gtk_button_new_with_label("I");
    GtkWidget* underlineBtn = gtk_button_new_with_label("U");

    gtk_box_append(GTK_BOX(toolbarHBox), newBtn);
    gtk_box_append(GTK_BOX(toolbarHBox), openBtn);
    gtk_box_append(GTK_BOX(toolbarHBox), saveBtn);
    gtk_box_append(GTK_BOX(toolbarHBox), gtk_separator_new(GTK_ORIENTATION_VERTICAL));
    gtk_box_append(GTK_BOX(toolbarHBox), boldBtn);
    gtk_box_append(GTK_BOX(toolbarHBox), italicBtn);
    gtk_box_append(GTK_BOX(toolbarHBox), underlineBtn);

    g_signal_connect(newBtn, "clicked", G_CALLBACK(newFile), NULL);
    g_signal_connect(openBtn, "clicked", G_CALLBACK(loadFile), NULL);
    g_signal_connect(saveBtn, "clicked", G_CALLBACK(saveCurrentFile), NULL);
    
    g_signal_connect(boldBtn, "clicked", G_CALLBACK(onBoldClicked), NULL);
    g_signal_connect(italicBtn, "clicked", G_CALLBACK(onItalicClicked), NULL);
    g_signal_connect(underlineBtn, "clicked", G_CALLBACK(onUnderlineClicked), NULL);

    editorView = gtk_text_view_new();
    gtk_text_view_set_wrap_mode(GTK_TEXT_VIEW(editorView), GTK_WRAP_WORD);

    GtkTextBuffer* buffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(editorView));

    gtk_text_buffer_create_tag(buffer, "kw_blue", "foreground", "blue", NULL);
    gtk_text_buffer_create_tag(buffer, "kw_green", "foreground", "darkgreen", NULL);
    gtk_text_buffer_create_tag(buffer, "kw_red", "foreground", "red", NULL);

    gtk_text_buffer_create_tag(buffer, "fmt_bold", "weight", PANGO_WEIGHT_BOLD, NULL);
    gtk_text_buffer_create_tag(buffer, "fmt_italic", "style", PANGO_STYLE_ITALIC, NULL);
    gtk_text_buffer_create_tag(buffer, "fmt_underline", "underline", PANGO_UNDERLINE_SINGLE, NULL);

    g_signal_connect(buffer, "changed", G_CALLBACK(+[](GtkTextBuffer*, gpointer) { applySyntaxHighlighting(); }), NULL);

    GtkWidget* scroll = gtk_scrolled_window_new();
    gtk_scrolled_window_set_child(GTK_SCROLLED_WINDOW(scroll), editorView);
    
    gtk_widget_set_vexpand(scroll, TRUE);
    gtk_box_append(GTK_BOX(mainVBox), scroll);

    gtk_window_present(GTK_WINDOW(mainWindow));
}

int main(int argc, char* argv[]) {
    GtkApplication* application = gtk_application_new("org.gtk.example.texteditor", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect(application, "activate", G_CALLBACK(onApplicationActivate), NULL);

    int status = g_application_run(G_APPLICATION(application), argc, argv);
    g_object_unref(application);

    return status;
}
